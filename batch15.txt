"use client";
import { useEffect, useState } from "react";
import { api } from "../lib/api";
import { useApi } from "../lib/useApi";
import { Badge, Button, Card, CardTitle, Skeleton } from "./ui";
import { CountdownClock } from "./CountdownClock";
interface IncidentRow {
id: string; title: string; severity: "LOW" | "MEDIUM" | "HIGH" | "CRITICAL";
status: string; detectedAt: string;
_count?: { notifications: number };
}
interface Obligation {
authority: string; deadlineAt: string; statute: string;
notes?: string; basis?: string; verified?: boolean;
}
const SEV_TONE = { CRITICAL: "fail", HIGH: "fail", MEDIUM: "warn", LOW: "zinc" } as const;
const IR_STATES = ["OPEN", "TRIAGE", "CONTAINED", "ERADICATED", "RECOVERED", "POSTMORTEM", "CLOSED"];
export function IncidentsClient() {
const { data, loading } = useApi<IncidentRow[]>("/v1/incidents");
const [selected, setSelected] = useState<string | null>(null);
const [obligations, setObligations] = useState<Obligation[] | null>(null);
const [computing, setComputing] = useState(false);
const [flash, setFlash] = useState<string | null>(null);
useEffect(() => { if (!selected && data?.length) setSelected(data[0].id); }, [data, selected]);
useEffect(() => {
if (!selected) return;
setObligations(null);
api.post<{ obligations: Obligation[] }>(`/v1/incidents/${selected}/obligations`)
.then((r) => setObligations(r.obligations ?? []))
.catch(() => setObligations([]));
}, [selected]);
const current = data?.find((i) => i.id === selected);
const stateIdx = current ? IR_STATES.indexOf(current.status) : -1;
const recompute = async () => {
if (!selected) return;
setComputing(true);
try {
const r = await api.post<{ obligations: Obligation[]; clocksStarted: number }>(
`/v1/incidents/${selected}/obligations`);
setObligations(r.obligations ?? []);
setFlash(`Recomputed ${r.obligations?.length ?? 0} obligations · ${r.clocksStarted} SLA clocks started`);
setTimeout(() => setFlash(null), 4000);
} finally { setComputing(false); }
};
if (loading) return <Skeleton className="h-96" />;
if (!data?.length) return (
<Card className="mx-auto max-w-lg p-10 text-center text-muted-foreground">No incidents. Clean record.</Card>
);
return (
<div className="mx-auto max-w-4xl space-y-4">
<h1 className="text-lg font-semibold">Incidents</h1>
{flash && <div className="rounded-md border border-info/30 bg-info/5 px-3 py-2 text-xs text-info">{flash}</div>}
<div className="grid gap-4 md:grid-cols-[280px_1fr]">
<Card>
<CardTitle>Open & recent</CardTitle>
<div className="divide-y divide-border">
{data.map((i) => (
<button key={i.id} onClick={() => setSelected(i.id)}
className={`flex w-full items-center gap-2 px-3 py-2.5 text-left text-xs ${selected === i.id ? "bg-muted font-medium" : "hover:bg-muted/40"}`}>
<span className="min-w-0 flex-1 truncate">{i.title}</span>
<Badge tone={SEV_TONE[i.severity]}>{i.severity}</Badge>
</button>
))}
</div>
</Card>
<div className="space-y-4">
{current && (
<Card>
<CardTitle right={<Badge tone="info">{current.status}</Badge>}>{current.title}</CardTitle>
<div className="space-y-3 p-4">
<div className="flex flex-wrap items-center gap-1.5">
{IR_STATES.map((s, i) => (
<span key={s}>
<span className={`rounded px-1.5 py-0.5 text-[10px] font-semibold ${i <= stateIdx ? "bg-info/10 text-info" : "bg-muted text-muted-foreground"}`}>{s}</span>
{i < IR_STATES.length - 1 && <span className="mx-0.5 text-muted-foreground">→</span>}
</span>
))}
</div>
<p className="text-xs text-muted-foreground">Detected {new Date(current.detectedAt).toLocaleString()}</p>
</div>
</Card>
)}
<Card>
<CardTitle right={<Button variant="outline" onClick={recompute} disabled={computing}>{computing ? "Computing…" : "Recompute clocks"}</Button>}>
Notification obligations
</CardTitle>
<div className="divide-y divide-border">
{obligations === null && <p className="p-6 text-center text-xs text-muted-foreground">Computing jurisdictional deadlines…</p>}
{obligations?.length === 0 && <p className="p-6 text-center text-xs text-muted-foreground">No obligations for the recorded facts. Set affected systems / data categories to compute clocks.</p>}
{obligations?.map((o, i) => (
<div key={i} className="flex items-center gap-3 px-4 py-3">
<div className="min-w-0 flex-1">
<p className="text-[13px] font-medium">{o.authority}</p>
<p className="truncate text-xs text-muted-foreground">{o.statute}{o.notes ? ` · ${o.notes}` : ""}</p>
</div>
{o.verified === false && <Badge tone="draft">provisional</Badge>}
<CountdownClock deadline={o.deadlineAt} />
</div>
))}
</div>
</Card>
</div>
</div>
</div>
);
}
