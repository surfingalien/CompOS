"use client";
import { useEffect, useMemo, useRef, useState } from "react";
import { useRouter } from "next/navigation";
import { Icon, Kbd } from "./ui";
export interface PaletteItem { id: string; label: string; group: string; icon: string; href?: string; action?: () => void }
export function CommandPalette({ open, onClose, items }:
{ open: boolean; onClose: () => void; items: PaletteItem[] }) {
const router = useRouter();
const [q, setQ] = useState("");
const [idx, setIdx] = useState(0);
const inputRef = useRef<HTMLInputElement>(null);
const filtered = useMemo(() => {
const s = q.toLowerCase();
return s ? items.filter((i) => i.label.toLowerCase().includes(s)) : items;
}, [q, items]);
useEffect(() => { if (open) { setQ(""); setIdx(0); setTimeout(() => inputRef.current?.focus(), 30); } }, [open]);
useEffect(() => setIdx(0), [q]);
if (!open) return null;
const run = (i: PaletteItem) => { onClose(); if (i.href) router.push(i.href); i.action?.(); };
return (
<div className="fixed inset-0 z-[60] flex items-start justify-center bg-zinc-950/40 p-4 pt-[14vh] backdrop-blur-[2px] animate-fade-in" onClick={onClose}>
<div className="w-full max-w-xl animate-scale-in overflow-hidden rounded-xl border border-border bg-raised shadow-pop" onClick={(e) => e.stopPropagation()}>
<div className="flex items-center gap-2.5 border-b border-border px-3.5">
<Icon name="search" className="h-4 w-4 text-muted-foreground" />
<input ref={inputRef} value={q} onChange={(e) => setQ(e.target.value)}
onKeyDown={(e) => {
if (e.key === "ArrowDown") { e.preventDefault(); setIdx((i) => Math.min(i + 1, filtered.length - 1)); }
if (e.key === "ArrowUp") { e.preventDefault(); setIdx((i) => Math.max(i - 1, 0)); }
if (e.key === "Enter" && filtered[idx]) run(filtered[idx]);
}}
placeholder="Search or jump to…"
className="h-12 flex-1 bg-transparent text-[13px] outline-none placeholder:text-muted-foreground/60" />
<Kbd>esc</Kbd>
</div>
<div className="max-h-[320px] overflow-y-auto p-1.5">
{filtered.length === 0 && <p className="px-3 py-6 text-center text-xs text-muted-foreground">No matches for “{q}”.</p>}
{filtered.map((i, n) => (
<button key={i.id} onClick={() => run(i)} onMouseEnter={() => setIdx(n)}
className={`flex w-full items-center gap-2.5 rounded-lg px-3 py-2 text-left text-[13px] ${n === idx ? "bg-muted" : ""}`}>
<Icon name={i.icon} className="h-4 w-4 text-muted-foreground" />
<span className="flex-1">{i.label}</span>
<span className="text-[10px] uppercase tracking-wide text-muted-foreground/70">{i.group}</span>
</button>
))}
</div>
</div>
</div>
);
}

import { useEffect, useState } from "react";
export function CountdownClock({ deadline }: { deadline: string }) {
const [now, setNow] = useState(Date.now());
useEffect(() => {
const t = setInterval(() => setNow(Date.now()), 1000);
return () => clearInterval(t);
}, []);
const ms = new Date(deadline).getTime() - now;
if (ms <= 0)
return <span className="rounded-md border border-fail/25 bg-fail/10 px-2 py-1 font-mono text-xs font-bold text-fail">OVERDUE</span>;
const h = Math.floor(ms / 3.6e6), m = Math.floor((ms % 3.6e6) / 6e4), s = Math.floor((ms % 6e4) / 1000);
const pct = Math.min(100, Math.max(0, 100 - (ms / 8.64e7) * 100));
const tone = ms < 7.2e6 ? "text-fail" : ms < 1.728e7 ? "text-warn" : "text-zinc-700 dark:text-zinc-200";
return (
<div>
<div className={`font-mono text-sm font-semibold tabular-nums ${tone} ${ms < 7.2e6 ? "animate-pulse" : ""}`}>
{h > 24 ? `${Math.floor(h / 24)}d ${h % 24}h` : `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`}
</div>
<div className="mt-1 h-1 w-24 overflow-hidden rounded-full bg-muted">
<div className={`h-full rounded-full ${ms < 7.2e6 ? "bg-fail" : ms < 1.728e7 ? "bg-warn" : "bg-pass"}`} style={{ width: `${pct}%` }} />
</div>
</div>
);
}
