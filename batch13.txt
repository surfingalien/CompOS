/**
* BATCH 13 — demo seed: every screen renders populated on first boot.
*
* pnpm db:seed:demo (after the base seeds; idempotent — safe to re-run)
*
* Creates for org "acme" / sarah@acme.test:
* • Controls: unified bootstrap (all four frameworks) + starter checks
* • Mock OKTA integration (plaintext-marker creds — seed needs no AWS/KMS)
* • Hash-chained evidence + TestRuns + quality grades on four controls
* • Tasks: open / in-progress / DONE-within-verify-window / auto-reopened
* • Vendors w/ certs (one expiring, one expired) + reviews
* • Incident with computed statutory notification clocks
* • Policies (approved + AI draft) · questionnaire w/ drafts · answer library
* • Privacy: 5 DSARs (one OVERDUE), data map, DPO settings
* • Trust center (published, slug "acme") + NDA template + signed grant
* • Reg radar changes · demo API key · Enterprise-trial entitlement
* • Brain: seeded events/flow/anomaly/health snapshot (best-effort)
*
* ALL imports are dynamic so non-AI paths boot without LLM keys.
*/

// Placeholders let import-time SDK construction pass; real AI features
// still need real keys in .env.
process.env.ANTHROPIC_API_KEY ??= "demo-placeholder";
process.env.OPENAI_API_KEY ??= "demo-placeholder";
const DAY = 86_400_000;
const daysAgo = (n: number) => new Date(Date.now() - n * DAY);
const daysFromNow = (n: number) => new Date(Date.now() + n * DAY);
const hoursAgo = (n: number) => new Date(Date.now() - n * 3_600_000);

const { prisma } = await import("../src/client");

employeeCount: 120, hqCountry: "US" } });

{ frameworkCodes: ["SOC2", "ISO27001", "HIPAA", "GDPR"] });

const checks = await seedStarterChecks(org.id);

let integration = await prisma.integration.findFirst({

if (!integration) {
integration = await prisma.integration.create({ data: {
orgId: org.id, type: "OKTA", status: "CONNECTED",

// Demo marker — real connects use storeCredentials() (KMS envelope).

where: { orgId_slug: { orgId: org.id, slug } } });
return prisma.testRun.create({ data: {

testId: `${org.id}:${CHECK_OF[slug]}`, status,
startedAt: hoursAgo(ago), finishedAt: hoursAgo(ago),

? `${matched}/${total} items matched`
: `${failing} item(s) failed predicate`,

stillActiveInIdp: true },
{ email: "m***@example.com", hoursSinceTermination: 55,
stillActiveInIdp: true }] } } });
