/**
* DEV-ONLY bootstrap login. The original batches created `createSession`

create: { name: "Acme Corp", slug: "acme", industry: "SAAS",
employeeCount: 120, hqCountry: "US" } });
const user = await prisma.user.upsert({

create: { email: "sarah@acme.test", name: "Sarah Chen" } });

import { FastifyInstance } from "fastify";
import { prisma } from "@db/client";
import { bootstrapOrgControls, routeGdprWorkflows, storeCredentials,
unifiedLeverage } from "@core";
import { seedStarterChecks, onIntegrationConnected } from "@skills";
import { startWorkflowSafe } from "../temporal";
import type { IntegrationType } from "@prisma/client";
interface OnboardBody {
frameworks: string[]; industry?: string; employeeCount?: number; hqCountry?: string;
integrations?: { type: IntegrationType; credentials: unknown; config?: object }[];
}
export async function onboardingHandlers(app: FastifyInstance) {
/** The wizard's single commit: profile → controls → checks → GDPR routing
* → integrations (sealed creds + evidence-sync workflows) → leverage. */
app.post("/v1/onboarding", async (req, reply) => {
const body = req.body as OnboardBody;
const orgId = req.org!.id;
if (!body.frameworks?.length)
return reply.code(400).send({ title: "Select at least one framework", status: 400 });
await prisma.organization.update({ where: { id: orgId }, data: {
industry: body.industry, employeeCount: body.employeeCount,
hqCountry: body.hqCountry } });
const boot = await bootstrapOrgControls(orgId, { frameworkCodes: body.frameworks });
const checks = await seedStarterChecks(orgId);
const gdpr = body.frameworks.includes("GDPR")
? await routeGdprWorkflows(orgId) : null;
const connected: { type: string; attached: string[] }[] = [];
for (const it of body.integrations ?? []) {
try {
const integrationId = await storeCredentials(orgId, it.type,
it.credentials, it.config ?? {});
const resolved = await onIntegrationConnected(orgId, integrationId, it.type);
const snapshots = resolved.specs.filter((s) => s.controlId)
.map((s) => ({ dataType: s.dataType, controlId: s.controlId!,
title: s.title, freshnessDays: s.freshnessDays, expectStable: s.expectStable }));
if (snapshots.length)
await startWorkflowSafe("evidenceSyncWorkflow",
[{ orgId, integrationId, snapshots }],
`evidence-sync-${orgId}-${integrationId}`);
connected.push({ type: it.type, attached: resolved.attached });
} catch (e) {
connected.push({ type: it.type, attached: [] }); // credential error surfaced in UI
}
}
const leverage = await unifiedLeverage(orgId, body.frameworks);
return { controls: boot.controls, checks: checks.created,
gdprChecklistItems: gdpr?.length ?? 0, connected, leverage };
});
}

import { FastifyInstance } from "fastify";
import { prisma } from "@db/client";
import { createPrivacyRequest, verifyPrivacyRequest, compileAccessExport,
executeErasure, extendDsarDeadline } from "@core";
/** PUBLIC — subject-facing intake + magic-link verification (no session). */
export async function privacyPublicRoutes(app: FastifyInstance) {
app.post("/v1/privacy/intake", async (req, reply) => {
const { orgSlug, type, email, details } = req.body as {
orgSlug: string; type: "ACCESS" | "RECTIFY" | "ERASE" | "RESTRICT" | "PORT" | "OBJECT";
email: string; details?: string };
const org = await prisma.organization.findUnique({ where: { slug: orgSlug } });
if (!org || !email?.includes("@"))
return reply.code(400).send({ title: "Invalid request", status: 400 });
const { id } = await createPrivacyRequest(org.id, { type, email, details });
return reply.code(201).send({ id,
message: "Verification link sent. Deadline: 30 days (Art. 12)." });
});
app.post("/v1/privacy/verify", async (req) => {
const { requestId, token } = req.body as { requestId: string; token: string };
return { verified: await verifyPrivacyRequest(requestId, token) };
});
}
/** ADMIN — inbox + actions (registered inside the eitherAuth plugin). */
export async function privacyHandlers(app: FastifyInstance) {
app.get("/v1/privacy-requests/:id", async (req, reply) => {
const { id } = req.params as { id: string };
const r = await prisma.privacyRequest.findFirst({
where: { id, orgId: req.org!.id },
include: { events: { orderBy: { at: "asc" } } } });
if (!r) return reply.code(404).send({ title: "Not found", status: 404 });
return { id: r.id, type: r.type, status: r.status,
dueAt: r.dueAt, extensionTo: r.extensionTo,
verifiedAt: r.verifiedAt, createdAt: r.createdAt,
emailHash: r.emailHash.slice(0, 12) + "…", // minimization — never the email
detailsSealed: !!r.detailsEnc,
events: r.events.map((e) => ({ kind: e.kind, at: e.at, details: e.details })) };
});
app.post("/v1/privacy-requests/:id/compile", async (req, reply) => {
const { id } = req.params as { id: string };
const r = await prisma.privacyRequest.findFirstOrThrow({
where: { id, orgId: req.org!.id } });
if (!["ACCESS", "PORT"].includes(r.type))
return reply.code(400).send({ title: "Compile applies to ACCESS/PORT only", status: 400 });
if (r.status !== "VERIFIED")
return reply.code(409).send({ title: `Status is ${r.status} — must be VERIFIED`, status: 409 });
return compileAccessExport(req.org!.id, id);
});
app.post("/v1/privacy-requests/:id/deliver", async (req, reply) => {
const { id } = req.params as { id: string };
const r = await prisma.privacyRequest.findFirstOrThrow({
where: { id, orgId: req.org!.id } });
if (!["REVIEW", "COMPILING", "VERIFIED"].includes(r.status))
return reply.code(409).send({ title: `Cannot deliver from ${r.status}`, status: 409 });
await prisma.privacyRequest.update({ where: { id }, data: { status: "DELIVERED" } });
await prisma.privacyRequestEvent.create({ data: { requestId: id, kind: "DELIVERED",
actorMembershipId: req.user?.id } });
return { status: "DELIVERED" };
});
app.post("/v1/privacy-requests/:id/erase", async (req, reply) => {
const { id } = req.params as { id: string };
const r = await prisma.privacyRequest.findFirstOrThrow({
where: { id, orgId: req.org!.id } });
if (r.type !== "ERASE")
return reply.code(400).send({ title: "Not an erasure request", status: 400 });
if (r.status !== "VERIFIED")
return reply.code(409).send({ title: `Status is ${r.status} — must be VERIFIED`, status: 409 });
const actions = await executeErasure(req.org!.id, id);
return { status: "DELIVERED", actions };
});
app.post("/v1/privacy-requests/:id/extend", async (req, reply) => {
const { id } = req.params as { id: string };
const { reason } = (req.body ?? {}) as { reason?: string };
const r = await prisma.privacyRequest.findFirstOrThrow({
where: { id, orgId: req.org!.id } });
try {
const extensionTo = await extendDsarDeadline(id, reason ?? "Complex request",
req.user?.id ?? "");
return { extensionTo };
} catch (e: any) {
return reply.code(409).send({ title: e.message, status: 409 });
}
});
}

// OVERRIDE (batch 12) — adds patch(), the new screen endpoints, and dev
// auto-login: on first 401 the client calls /v1/auth/dev-login once and
// retries (hard-disabled in production on the server side).
import type { ChainProof, ControlDetail, QuestionnaireReviewData, ReadinessResponse } from "./types";
export const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? "";
export class ApiError extends Error {
constructor(public status: number, public body: unknown) { super(`API ${status}`); }
}
async function devLogin(): Promise<boolean> {
try {
const res = await fetch(`${API_BASE}/v1/auth/dev-login`,
{ method: "POST", credentials: "include" });
return res.ok;
} catch { return false; }
}
async function doFetch(method: string, path: string, body?: unknown) {
return fetch(`${API_BASE}${path}`, {
method,
headers: body ? { "content-type": "application/json" } : undefined,
body: body ? JSON.stringify(body) : undefined,
credentials: "include",
});
}
async function req<T>(method: string, path: string, body?: unknown): Promise<T> {
let res = await doFetch(method, path, body);
if (res.status === 401) {
if (await devLogin()) res = await doFetch(method, path, body);
}
if (!res.ok) throw new ApiError(res.status, await res.json().catch(() => ({})));
return (res.status === 204 ? undefined : await res.json()) as T;
}
export const api = {
get: <T>(p: string) => req<T>("GET", p),

"use client";
import Link from "next/link";
import { useEffect, useMemo, useState } from "react";
import { api } from "../lib/api";
import { useApi } from "../lib/useApi";
import { relTime } from "../lib/format";
import { Badge, Button, Card, CardTitle, Skeleton } from "./ui";
interface Task {
id: string; title: string; description: string | null;
status: "OPEN" | "IN_PROGRESS" | "BLOCKED" | "DONE" | "CANCELLED";
priority: "LOW" | "MEDIUM" | "HIGH" | "CRITICAL";
source: string; relatedType: string | null; relatedId: string | null;
dueAt: string | null; completedAt: string | null; createdAt: string;
}
const P_TONE: Record<string, "fail" | "warn" | "info" | "zinc"> =
{ CRITICAL: "fail", HIGH: "warn", MEDIUM: "info", LOW: "zinc" };
const SRC_LABEL: Record<string, string> = {
GAP_ANALYSIS: "gap", TEST_FAILURE: "test", REG_CHANGE: "reg change",
INCIDENT: "incident", VENDOR_REVIEW: "vendor", POLICY_ACK: "policy", MANUAL: "manual" };
const VERIFY_WINDOW = 24 * 3_600_000; // matches the workers' hourly sweep window
export function TasksClient() {
const { data, loading, refresh } = useApi<Task[]>("/v1/tasks");
const [busy, setBusy] = useState<string | null>(null);
const [flash, setFlash] = useState<string | null>(null);
// Poll every 30s: the remediation verification sweep reopens failed "DONE"
// tasks within the hour — the UI must surface that without a reload.
useEffect(() => {
const t = setInterval(refresh, 30_000);
return () => clearInterval(t);
}, [refresh]);
const act = async (id: string, status: string, note?: string) => {
setBusy(id);
try {
await api.taskUpdate(id, { status });
setFlash(note ?? null);
refresh();
} catch (e) { setFlash(`Update failed: ${(e as Error).message}`); }
finally { setBusy(null); }
};
const groups = useMemo(() => {
const open = (data ?? []).filter((t) => t.status !== "DONE" && t.status !== "CANCELLED");
const done = (data ?? []).filter((t) => t.status === "DONE");
return { open: open.sort(byPriority), done };
}, [data]);
if (loading) return <Skeleton className="h-96" />;
return (
<div className="mx-auto max-w-5xl space-y-4">
<div className="flex items-baseline justify-between">
<h1 className="text-lg font-semibold">Tasks</h1>
<span className="text-xs text-muted-foreground">{groups.open.length} open · verification sweep runs hourly</span>
</div>
{flash && <div className="rounded-md border border-info/30 bg-info/5 px-3 py-2 text-xs text-info">{flash}</div>}
<Card>
<CardTitle right={<Badge tone="zinc">{groups.open.length}</Badge>}>Open</CardTitle>
{groups.open.length === 0 && <Empty msg="No open tasks — clean program." />}
{groups.open.map((t) => (
<Row key={t.id} t={t} busy={busy === t.id}
onStart={() => act(t.id, "IN_PROGRESS")}
onDone={() => act(t.id, "DONE", "Marked done — auto-verification pending (hourly sweep re-runs the check).")} />
))}
</Card>
<Card>
<CardTitle right={<Badge tone="pass">{groups.done.length}</Badge>}>Completed — auto-verification</CardTitle>
{groups.done.length === 0 && <Empty msg="Nothing completed yet." />}
{groups.done.map((t) => <DoneRow key={t.id} t={t} />)}
</Card>
</div>
);
}
function byPriority(a: Task, b: Task) {
const o = { CRITICAL: 0, HIGH: 1, MEDIUM: 2, LOW: 3 };
return (o[a.priority] - o[b.priority]);
}
function Row({ t, busy, onStart, onDone }: { t: Task; busy: boolean;
onStart: () => void; onDone: () => void }) {
const overdue = t.dueAt && new Date(t.dueAt) < new Date();
const reopened = t.description?.includes("AUTO-REOPENED");
return (
<div className="flex items-center gap-3 border-b border-border px-4 py-3 last:border-0">
<Badge tone={P_TONE[t.priority]}>{t.priority}</Badge>
<div className="min-w-0 flex-1">
<p className="truncate text-[13px] font-medium">{t.title}</p>
<p className="text-xs text-muted-foreground">
{SRC_LABEL[t.source] ?? t.source}{t.dueAt && <> · due {relTime(t.dueAt)}</>}
{reopened && <span className="ml-2 font-medium text-fail">⚠ auto-verify failed — reopened</span>}
</p>
</div>
{overdue && <Badge tone="fail">overdue</Badge>}
{t.status === "OPEN" && <Button variant="outline" disabled={busy} onClick={onStart}>Start</Button>}
<Button variant="success" disabled={busy} onClick={onDone}
title="The hourly verification sweep re-runs the related check. If it fails, this task reopens automatically.">
Mark done ✓
</Button>
</div>
);
}
function DoneRow({ t }: { t: Task }) {
const within = t.completedAt && Date.now() - new Date(t.completedAt).getTime() < VERIFY_WINDOW;
return (

);

<Button variant="danger" onClick={() => run(() => api.privacyErase(selected.id), "Erasure executed — legal holds applied, Art. 19 notices queued.")} disabled={busy}>Execute erasure</Button>)}
{selected.status === "REVIEW" && (
<Button variant="success" onClick={() => run(() => api.privacyDeliver(selected.id), "Delivered — subject notified.")} disabled={busy}>Deliver to subject</Button>)}
{!selected.extensionTo && !["DELIVERED", "CLOSED_REJECTED", "CLOSED_NO_DATA"].includes(selected.status) && (
<Button variant="outline" onClick={() => setExtendOpen(true)}>Extend +2 months (once)</Button>)}
</div>
{/* timeline */}
<div>
<p className="mb-1 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Timeline</p>
<ol className="space-y-1.5 border-l border-border pl-3">
{selected.events.map((e, i) => (
<li key={i} className="text-xs">
<span className="font-medium">{e.kind}</span>
<span className="ml-2 text-muted-foreground">{new Date(e.at).toLocaleString()}</span>
{e.details && <p className="text-muted-foreground">{e.details.slice(0, 200)}</p>}
</li>
))}
</ol>
</div>
</div>
)}
</Modal>
<Modal open={extendOpen} onClose={() => setExtendOpen(false)} title="Extend deadline (Art. 12(3))">
<div className="space-y-2">
<p className="text-xs text-muted-foreground">One extension of up to two further months. The subject is informed within the original month. This cannot be applied twice.</p>
<input placeholder="Reason (complexity, volume…)" className="w-full rounded-md border border-border p-2 text-xs"
value={reason} onChange={(e) => setReason(e.target.value)} />
<div className="flex justify-end gap-2">
<Button variant="ghost" onClick={() => setExtendOpen(false)}>Cancel</Button>
<Button disabled={busy || reason.length < 5} onClick={() => {
if (!selected) return;
run(() => api.privacyExtend(selected.id, reason).then(() => { setExtendOpen(false; }), "Extended — new deadline recorded."));
}}>{busy ? "Extending…" : "Extend"}</Button>
</div>
</div>
</Modal>
</div>
);
}
