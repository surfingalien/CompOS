import { FastifyInstance } from "fastify";
import { prisma } from "@db/client";
import { brainDb } from "@brain";
/** RIA read-model for the /ria dashboard. Defensive: returns [] if the RIA
* tables aren't present yet, so the screen degrades instead of 500ing. */
async function safe<T>(fn: () => Promise<T>, fallback: T): Promise<T> {
try { return await fn(); } catch { return fallback; }
}
export async function riaHandlers(app: FastifyInstance) {
app.get("/v1/ria/jurisdictions", async () =>
safe(() => prisma.jurisdiction.findMany({ orderBy: { code: "asc" } }), [] as any[]));
app.get("/v1/ria/sources", async () => {
const [sources, jurisdictions] = await Promise.all([
safe(() => prisma.regSourceCountry.findMany({ orderBy: { reliability: "desc" } }), [] as any[]),
safe(() => prisma.jurisdiction.findMany(), [] as any[]),
]);
const byId = new Map(jurisdictions.map((j: any) => [j.id, j.code]));
return sources.map((s: any) => ({
id: s.id, name: s.name, url: s.url, parserType: s.parserType,
jurisdictionCode: byId.get(s.jurisdictionId) ?? null,
reliability: s.reliability ?? 0.5, status: s.status,
lastCheckedAt: s.lastCheckedAt, docsExtracted: s.docsExtracted ?? 0,
crawlCount: s.crawlCount ?? 0, discoveredBy: s.discoveredBy,
}));
});
app.get("/v1/ria/graph", async (req) => {
const { jurisdiction } = req.query as { jurisdiction?: string };
const nodes = await safe(() => prisma.regKnowledgeNode.findMany({
where: jurisdiction ? { jurisdictionId: jurisdiction } : {},
orderBy: { confidence: "desc" }, take: 500 }), [] as any[]);
const jurisdictions = await safe(() => prisma.jurisdiction.findMany(), [] as any[]);
const byId = new Map(jurisdictions.map((j: any) => [j.id, j.code]));
const counts: Record<string, number> = { REGULATION: 0, REQUIREMENT: 0, CONTROL: 0, GUIDANCE: 0 };
for (const n of nodes) if (counts[n.nodeType] !== undefined) counts[n.nodeType]++;
return { counts, nodes: nodes.map((n: any) => ({
id: n.id, nodeType: n.nodeType, refCode: n.refCode, title: n.title,
jurisdictionCode: byId.get(n.jurisdictionId) ?? null,
confidence: n.confidence, verified: n.verified,
linkedRequirementId: n.linkedRequirementId, linkedControlTemplateId: n.linkedControlTemplateId })) };
});
app.get("/v1/ria/drift", async () =>
safe(() => brainDb.anomaly.findMany({
where: { detector: "ria-drift", status: { in: ["OPEN", "ACKNOWLEDGED"] } },
orderBy: { at: "desc" }, take: 50 }), [] as any[]));
}

import { brainDb } from "@brain";
import { calibrate } from "@questionnaire";
import { ReviewSource, RenderBundle, enqueue } from "./types";
function credentialFor(jurisdiction: string | null): string {
if (!jurisdiction) return "CIPA";
if (jurisdiction === "EU" || jurisdiction.startsWith("EU-") || jurisdiction === "GB") return "LEGAL_EU";
if (jurisdiction.startsWith("US")) return "LEGAL_US_STATES";
return "CIPA";
}
/** RIA_DRIFT queue: extraction-quality drift anomalies surfaced for expert
* adjudication. Approve → recalibrate retrieval + resolve. Reject → mark
* FALSE_POSITIVE so the detector won't re-fire. */
export const riaDriftSource: ReviewSource = {
queueType: "RIA_DRIFT" as never,
async scanAndEnqueue() {
const open = await brainDb.anomaly.findMany({
where: { detector: "ria-drift", status: "OPEN" } });
let enqueued = 0;
for (const a of open) {
const d = (a.details ?? {}) as { jurisdiction?: string };
await enqueue({
queueType: "RIA_DRIFT" as never, sourceId: a.id,
title: `RIA drift: ${a.subsystem}/${a.metric} = ${a.value} (baseline ${a.baseline})`,
jurisdiction: d.jurisdiction ?? null,
priority: a.severity === "CRITICAL" ? "CRITICAL" : "HIGH",
confidence: 1, requiredCredential: credentialFor(d.jurisdiction ?? null),
preview: { metric: a.metric, value: a.value, baseline: a.baseline,
subsystem: a.subsystem, jurisdiction: d.jurisdiction ?? null },
});
enqueued++;
}
return { enqueued };
},
async render(id): Promise<RenderBundle> {
const a = await brainDb.anomaly.findUniqueOrThrow({ where: { id } });
const d = (a.details ?? {}) as Record<string, unknown>;
return {
header: { title: `RIA extraction drift — ${a.subsystem}/${a.metric}`,
jurisdiction: (d.jurisdiction as string) ?? undefined,
source: "Brain ria-drift detector" },
body: [
{ kind: "evidence-quote", quotes: [{ source: "ria-drift detector",
text: `value ${a.value} vs baseline ${a.baseline}` +
(a.zScore != null ? ` (z=${a.zScore})` : "") }] },
{ kind: "impact-list", items: [
{ label: `Subsystem: ${a.subsystem}`, level: a.severity },
{ label: "Approving recalibrates retrieval thresholds for the affected org",
level: "INFO", action: "Recalibrate on approval" },
{ label: "Rejecting marks the anomaly FALSE_POSITIVE", level: "INFO" },
] },
],
risk: a.severity === "CRITICAL"
? ["CRITICAL drift — auto-clone quality is degrading; adjudicate before more answers ship."] : [],
};
},
async approve(id, _reviewer, _note) {
const a = await brainDb.anomaly.findUnique({ where: { id } });
if (a?.orgId) await calibrate(a.orgId).catch(() => {});
await brainDb.anomaly.update({ where: { id }, data: { status: "RESOLVED" } });
return { applied: "RIA_RECALIBRATED" };
},
async reject(id, _reviewer, _note) {
await brainDb.anomaly.update({ where: { id }, data: { status: "FALSE_POSITIVE" } });
},
};

// OVERRIDE — registers the RIA_DRIFT source alongside the other three queues.
import { prisma } from "@db/client";
import { emit } from "@brain/emit";
import type { ReviewerCredential } from "@prisma/client";
import { frameworkContentSource } from "./sources/frameworkContent";
import { matrixProposalSource } from "./sources/matrixProposal";
import { regChangeSource } from "./sources/regChange";
import { riaDriftSource } from "./sources/riaDrift";
import type { ReviewSource } from "./sources/types";
export const SOURCES: Record<string, ReviewSource> = {
FRAMEWORK_CONTENT: frameworkContentSource,
MATRIX_PROPOSAL: matrixProposalSource,
REG_CHANGE: regChangeSource,
RIA_DRIFT: riaDriftSource,
};
export class Conflict extends Error {}
export class NotFoundError extends Error {}
async function record(itemId: string, actorMembershipId: string | null,
action: string, note?: string) {
await prisma.reviewDecision.create({ data: { reviewItemId: itemId,
actorMembershipId: actorMembershipId ?? "", action, note } });
}
async function assertCredential(membershipId: string, credential: ReviewerCredential | null) {
if (!credential) return;
const profile = await prisma.reviewerProfile.findUnique({ where: { membershipId } });
if (!profile?.active || !profile.credentials.includes(credential))
throw new Conflict(`This item requires the ${credential} credential`);
}
async function finalize(itemId: string, reviewerMembershipId: string | null,
status: string, note?: string) {
await prisma.reviewItem.update({ where: { id: itemId },
data: { status: status as never, decidedByMembershipId: reviewerMembershipId ?? "",
decidedAt: new Date(), decisionNote: note, claimedByMembershipId: null } });
await record(itemId, reviewerMembershipId, status, note);
}
export async function scanAllQueues() {
const results = await Promise.all(Object.values(SOURCES).map((s) => s.scanAndEnqueue()));
return { enqueued: results.reduce((s, r) => s + r.enqueued, 0) };
}
export async function claim(itemId: string, reviewerMembershipId: string) {
const item = await prisma.reviewItem.findUniqueOrThrow({ where: { id: itemId } });
if (item.status !== "PENDING") throw new Conflict("Item not claimable");
await assertCredential(reviewerMembershipId, item.requiredCredential);
await prisma.reviewItem.update({ where: { id: itemId },
data: { status: "CLAIMED", claimedByMembershipId: reviewerMembershipId, claimedAt: new Date() } });
await record(itemId, reviewerMembershipId, "CLAIM");
return item;
}
export async function decide(itemId: string, reviewerMembershipId: string,
decision: "APPROVE" | "REJECT" | "DEFER", note: string, deferUntil?: Date) {
const item = await prisma.reviewItem.findUniqueOrThrow({ where: { id: itemId } });
await assertCredential(reviewerMembershipId, item.requiredCredential);
const source = SOURCES[item.queueType];
if (!source) throw new NotFoundError(`No source for queue ${item.queueType}`);
if (decision === "DEFER") {
await prisma.reviewItem.update({ where: { id: itemId },
data: { status: "PENDING", deferredUntil: deferUntil, claimedByMembershipId: null } });
await record(itemId, reviewerMembershipId, "DEFER", note);
return { status: "DEFERRED" };
}
if (decision === "REJECT") {
await source.reject(item.sourceId, reviewerMembershipId, note);
await finalize(itemId, reviewerMembershipId, "REJECTED", note);
return { status: "REJECTED" };
}
if (item.priority === "CRITICAL") {
if (!item.firstApprovalBy) {
await prisma.reviewItem.update({ where: { id: itemId },
data: { status: "APPROVED_PENDING_SECOND", firstApprovalBy: reviewerMembershipId,
firstApprovalAt: new Date() } });
await record(itemId, reviewerMembershipId, "APPROVE_FIRST", note);
return { status: "APPROVED_PENDING_SECOND" };
}
if (item.firstApprovalBy === reviewerMembershipId)
throw new Conflict("Four-eyes: second approval must come from a different reviewer");
}
const result = await source.approve(item.sourceId, reviewerMembershipId, note, !!item.firstApprovalBy);
await finalize(itemId, reviewerMembershipId, "APPROVED", note);
return result ?? { status: "APPROVED" };
}
export async function releaseClaim(itemId: string, reviewerMembershipId: string) {
await prisma.reviewItem.update({ where: { id: itemId },
data: { status: "PENDING", claimedByMembershipId: null } });
await record(itemId, reviewerMembershipId, "RELEASE");
}

method,
headers: body ? { "content-type": "application/json" } : undefined,
body: body ? JSON.stringify(body) : undefined,
credentials: "include",
});

const r: any = await api.reviewDecide(selectedId, decision, note);
say(r?.status === "APPROVED_PENDING_SECOND"
? "First approval recorded — awaiting a second reviewer (four-eyes)."
: `${decision === "APPROVE" ? "Approved — applied downstream" : decision === "REJECT" ? "Rejected" : "Deferred"}.`);
setSelectedId(null); setDetail(null); setNote("");
queue.refresh(); stats.refresh();
} catch (e: any) { say(e.body?.title ?? e.message); }
finally { setBusy(false); }
};
return (
<div className="space-y-4">
<div className="flex flex-wrap items-center gap-3">
<h1 className="text-lg font-semibold">Review Console — Expert Gate</h1>
<Badge tone="zinc">{(queue.data ?? []).length} queued</Badge>
{(stats.data?.slaBreaches ?? 0) > 0 && <Badge tone="fail">{stats.data!.slaBreaches} SLA breaches</Badge>}
<div className="flex-1" />
<Button variant="outline" onClick={async () => { setBusy(true); try { const r = await api.reviewScan(); say(`Scan complete — ${r.enqueued} item(s) enqueued`); queue.refresh(); } finally { setBusy(false); } }} disabled={busy}>
{busy ? "Scanning…" : "Scan queues"}
</Button>
</div>
{flash && <div className="rounded-md border border-info/30 bg-info/5 px-3 py-2 text-xs text-info">{flash}</div>}
<Card>
<Tabs<Tab> active={tab} onChange={setTab} tabs={[
{ id: "ALL", label: "All", count: queue.data?.length ?? 0 },
{ id: "FRAMEWORK_CONTENT", label: "Framework", count: (queue.data ?? []).filter((i) => i.queueType === "FRAMEWORK_CONTENT").length },
{ id: "MATRIX_PROPOSAL", label: "Matrix", count: (queue.data ?? []).filter((i) => i.queueType === "MATRIX_PROPOSAL").length },
{ id: "REG_CHANGE", label: "Reg change", count: (queue.data ?? []).filter((i) => i.queueType === "REG_CHANGE").length },
{ id: "RIA_DRIFT", label: "RIA drift", count: (queue.data ?? []).filter((i) => i.queueType === "RIA_DRIFT").length },
]} />
<div className="grid flex-1 grid-cols-1 overflow-hidden lg:grid-cols-[360px_1fr]">
<div className="max-h-[70vh] overflow-y-auto border-r border-border">
{queue.loading ? <div className="p-4"><Skeleton /></div> : items.map((it) => (
<button key={it.id} onClick={() => setSelectedId(it.id)}
className={`flex w-full flex-col gap-1 border-b border-border px-3 py-2.5 text-left ${selectedId === it.id ? "bg-muted" : "hover:bg-muted/50"}`}>
<div className="flex items-center gap-2">
<Badge tone={Q_TONE[it.queueType] ?? "zinc"}>{it.queueType}</Badge>
<Badge tone={it.priority === "CRITICAL" ? "fail" : it.priority === "HIGH" ? "warn" : "zinc"}>{it.priority}</Badge>
<span className="ml-auto text-[11px] text-muted-foreground">{it.status}</span>
</div>
<span className="line-clamp-2 text-[13px] font-medium">{it.title}</span>
</button>
))}
{items.length === 0 && !queue.loading && <p className="p-8 text-center text-xs text-muted-foreground">Queue clear for this filter.</p>}
</div>
<div className="max-h-[70vh] overflow-y-auto p-5">
{!selectedId && <p className="py-10 text-center text-xs text-muted-foreground">Select an item to render its review bundle.</p>}
{selectedId && !detail && <Skeleton className="h-64" />}
{detail && (
<div className="space-y-4">
<div>
<div className="flex items-center gap-2">
<Badge tone={Q_TONE[detail.item.queueType] ?? "zinc"}>{detail.item.queueType}</Badge>
{detail.item.jurisdiction && <Badge tone="info">{detail.item.jurisdiction}</Badge>}
{detail.item.requiredCredential && <Badge tone="draft">requires {detail.item.requiredCredential}</Badge>}
</div>
<h2 className="mt-2 text-[15px] font-semibold leading-relaxed">{detail.bundle.header.title}</h2>
<p className="mt-1 text-xs text-muted-foreground">Source: {detail.bundle.header.source}</p>
</div>
{detail.bundle.risk?.length > 0 && (
<div className="rounded-md border border-warn/30 bg-warn/5 p-3">
<p className="text-xs font-medium text-warn">Reviewer caution</p>
<ul className="mt-1 list-disc pl-4 text-xs text-zinc-700">{detail.bundle.risk.map((r: string, i: number) => <li key={i}>{r}</li>)}</ul>
</div>
)}
{detail.bundle.body?.map((b: any, i: number) => <BundleBlock key={i} b={b} />)}
<textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2}
placeholder="Decision note (recorded in the audit trail)…"
className="w-full rounded-md border border-border p-2 text-[13px]" />
<div className="flex flex-wrap gap-2">
<Button variant="success" onClick={() => act("APPROVE")} disabled={busy}>Approve</Button>
<Button variant="danger" onClick={() => act("REJECT")} disabled={busy}>Reject</Button>
<Button variant="outline" onClick={() => act("DEFER")} disabled={busy}>Defer</Button>
</div>
</div>
)}
</div>
</div>
</Card>
</div>
);
}
function BundleBlock({ b }: { b: any }) {
switch (b.kind) {
case "diff-table":
return (<Card><CardTitle>Proposed change (diff)</CardTitle>
<div className="overflow-x-auto"><table className="w-full text-[13px]"><thead><tr className="border-b border-border text-left text-[11px] uppercase text-muted-foreground">
<th className="px-3 py-2">Field</th><th className="px-3 py-2">From</th><th className="px-3 py-2">To</th><th className="px-3 py-2">Conf</th></tr></thead>
<tbody className="divide-y divide-border">{b.rows.map((r: any, i: number) => (
<tr key={i}><td className="px-3 py-2 font-mono text-xs">{r.field}</td>
<td className="px-3 py-2 text-xs text-fail line-through">{r.from}</td>
<td className="px-3 py-2 text-xs text-pass">{r.to}</td>
<td className="px-3 py-2 text-xs tabular-nums">{r.confidence?.toFixed(2) ?? "—"}</td></tr>))}</tbody></table></div></Card>);
case "paraphrase-card":
return (<Card><CardTitle>Requirement paraphrase</CardTitle><div className="p-3">
<p className="font-mono text-xs text-info">{b.requirement.code}</p>
<p className="mt-1 text-[13px] font-medium">{b.requirement.title}</p>
<p className="mt-1 text-xs leading-relaxed text-muted-foreground">{b.requirement.desc}</p>
{b.officialHint && <p className="mt-2 text-[11px] text-warn">{b.officialHint}</p>}</div></Card>);
case "impact-list":
return (<Card><CardTitle>Impact assessment</CardTitle><div className="divide-y divide-border">
{b.items.map((it: any, i: number) => (
<div key={i} className="flex items-center gap-2 px-3 py-2 text-[13px]">
<Badge tone={it.level === "CRITICAL" ? "fail" : it.level === "MAJOR" ? "warn" : "zinc"}>{it.level}</Badge>
<span className="flex-1">{it.label}</span>{it.action && <span className="text-xs text-muted-foreground">{it.action}</span>}</div>))}</div></Card>);
case "evidence-quote":
return (<Card><CardTitle>Extracted evidence</CardTitle><div className="space-y-2 p-3">
{b.quotes.map((q: any, i: number) => (
<div key={i} className="border-l-2 border-border pl-3"><p className="text-[11px] font-medium text-muted-foreground">{q.source}</p>
<p className="mt-0.5 text-xs leading-relaxed text-zinc-700">{q.text}</p></div>))}</div></Card>);
case "affected-orgs":
return (<Card><CardTitle>Blast radius</CardTitle><div className="p-3 text-[13px]">
<p><span className="font-semibold tabular-nums">{b.count}</span> org(s) affected.</p>
{b.orgs?.length > 0 && <p className="mt-1 font-mono text-[11px] text-muted-foreground">{b.orgs.join(", ")}</p>}</div></Card>);
default: return null;
}
}
