interface CompletionRequest {
system: string;
messages: { role: "user" | "assistant"; content: string }[];
maxTokens: number;
temperature: number;
jsonMode: boolean;
}
interface CompletionResult {
text: string; promptTokens: number; completionTokens: number;
model: string; provider: "anthropic" | "openai";
}
let _anthropic: Anthropic | null = null;
let _openai: OpenAI | null = null;
function anthropicClient(): Anthropic { return (_anthropic ??= new Anthropic()); }
function openaiClient(): OpenAI { return (_openai ??= new OpenAI()); }
const PRICING: Record<string, { in: number; out: number }> = {
"claude-sonnet-4-20250514": { in: 3 / 1e6, out: 15 / 1e6 },
"claude-3-5-haiku-20241022": { in: 0.8 / 1e6, out: 4 / 1e6 },
"gpt-4o": { in: 2.5 / 1e6, out: 10 / 1e6 },
};
async function complete(req: CompletionRequest): Promise<CompletionResult> {
const primary = process.env.LLM_PRIMARY ?? "anthropic";
try {
return primary === "anthropic" ? await anthropicComplete(req) : await openaiComplete(req);
} catch (err) {
if (process.env.LLM_FALLBACK === "off") throw err;
return primary === "anthropic" ? openaiComplete(req) : anthropicComplete(req);
}
}
async function anthropicComplete(req: CompletionRequest): Promise<CompletionResult> {
const model = req.maxTokens > 2000 ? "claude-sonnet-4-20250514" : "claude-3-5-haiku-20241022";
const res = await anthropicClient().messages.create({
model, max_tokens: req.maxTokens, temperature: req.temperature, system: req.system,
messages: req.messages.map((m) => ({ role: m.role, content: m.content })),
});
const text = res.content.filter((b) => b.type === "text").map((b) => b.text).join("");
return { text, model, provider: "anthropic",
promptTokens: res.usage.input_tokens, completionTokens: res.usage.output_tokens };
}
async function openaiComplete(req: CompletionRequest): Promise<CompletionResult> {
const res = await openaiClient().chat.completions.create({
model: "gpt-4o", max_tokens: req.maxTokens, temperature: req.temperature,
...(req.jsonMode ? { response_format: { type: "json_object" as const } } : {}),
messages: [{ role: "system", content: req.system }, ...req.messages],
});
return { text: res.choices[0]?.message?.content ?? "",
model: "gpt-4o", provider: "openai",
promptTokens: res.usage?.prompt_tokens ?? 0,
completionTokens: res.usage?.completion_tokens ?? 0 };
}

const p = PRICING[model] ?? { in: 3 / 1e6, out: 15 / 1e6 };
return pt * p.in + ct * p.out;
}

const PROMPT_DIR = join(__dirname, "prompts");
async function resolvePrompt(key: string, fallback: string): Promise<{ system: string; version: number }> {
const row = await prisma.promptTemplate.findFirst({
where: { key, status: "ACTIVE" }, orderBy: { version: "desc" } });
if (row) return { system: row.systemPrompt, version: row.version };
const file = join(PROMPT_DIR, `${key}.md`);
if (existsSync(file)) return { system: readFileSync(file, "utf8"), version: 0 };
return { system: fallback, version: 0 };
}

// ── THE ENTRY POINT every AI feature calls ──────────────────────
export async function runAgent<T = unknown>(
name: string, input: unknown, opts?: { orgId?: string },
): Promise<T> {
const def = AGENTS[name];
if (!def) throw new Error(`Unknown agent: ${name}`);
const { system, version } = await resolvePrompt(def.promptKey, def.fallbackSystem);
const userContent = typeof input === "string" ? input : JSON.stringify(input, null, 2);
const job = await prisma.generationJob.create({
data: { orgId: opts?.orgId ?? "platform", promptKey: def.promptKey,
promptVersion: version, model: def.model,
inputContext: input as object, status: "RUNNING" } });
const t0 = Date.now();
try {
let result = await complete({ system,
messages: [{ role: "user", content: userContent }],
maxTokens: def.maxTokens, temperature: def.model === "haiku" ? 0.1 : 0.3,
jsonMode: true });
let parsed = tryParseJson(result.text);
if (parsed === null) { // one repair retry
const repaired = await complete({ system,
messages: [
{ role: "user", content: userContent },
{ role: "assistant", content: result.text },
{ role: "user", content: "Your previous output was not valid JSON. Output ONLY valid JSON matching the schema, no prose, no markdown fences." },
],
maxTokens: def.maxTokens, temperature: 0, jsonMode: true });
parsed = tryParseJson(repaired.text);
result = { ...repaired,
promptTokens: result.promptTokens + repaired.promptTokens,
completionTokens: result.completionTokens + repaired.completionTokens };
}
if (parsed === null) throw new Error("Agent failed to produce valid JSON after repair retry");
let output = parsed;
if (def.redact?.length) {
let json = JSON.stringify(parsed);
for (const re of def.redact) json = json.replace(re, "[REDACTED]");
output = JSON.parse(json);
}
const cost = costUsd(result.model, result.promptTokens, result.completionTokens);
if (cost > def.budget.maxUsdPerRun)
console.warn(`[agents] ${name} exceeded budget: $${cost.toFixed(4)}`);
await prisma.generationJob.update({ where: { id: job.id },
data: { output, status: "SUCCEEDED", completedAt: new Date(),
promptTokens: result.promptTokens, completionTokens: result.completionTokens,
costUsd: cost, latencyMs: Date.now() - t0, model: result.model } });
const { emit } = await import("@brain/emit");
emit({ type: "llm.generation.completed", source: "agent", orgId: opts?.orgId,
payload: { agent: name, costUsd: cost, latencyMs: Date.now() - t0 } });
return output as T;
} catch (err) {
await prisma.generationJob.update({ where: { id: job.id },
data: { status: "FAILED", error: String(err).slice(0, 500), completedAt: new Date() } });
const { emit } = await import("@brain/emit");
emit({ type: "llm.generation.failed", source: "agent", orgId: opts?.orgId,
severityOverride: "ERROR", payload: { agent: name, error: String(err).slice(0, 200) } });
throw err;
}
}
function tryParseJson(text: string): unknown | null {
const cleaned = text.replace(/^```(?:json)?\s*/m, "").replace(/```\s*$/m, "").trim();
try { return JSON.parse(cleaned); } catch { return null; }
}
