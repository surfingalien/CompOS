// Batch 16 — platform observability + regulatory-governance surfaces.
// Org-scoped reads go through rlsClient (RLS binds); platform/global tables
// (StateBreachRule, FrameworkRequirement, RegChange, ControlTemplate, …) have
// no orgId and are read with plain prisma.
import { FastifyInstance } from "fastify";
import { prisma, rlsClient } from "@db/client";
import { controlTestState, suggestMappingsForNewFramework } from "@core";
import { connectorFor } from "@skills";
const CONNECTOR_TYPES = ["OKTA", "AWS", "GITHUB", "BAMBOOHR"] as const;
export async function platformHandlers(app: FastifyInstance) {
// ── EVIDENCE LIBRARY ─────────────────────────────────────────────
app.get("/v1/evidence-library", async (req) => {
const { controlSlug, sourceType, limit } = req.query as
{ controlSlug?: string; sourceType?: string; limit?: string };
return rlsClient().asOrg(req.org!.id, async (tx) => {
const rows = await tx.evidence.findMany({
where: {
...(controlSlug ? { control: { slug: controlSlug } } : {}),
...(sourceType ? { sourceType: sourceType as never } : {}),
},
include: {
control: { select: { slug: true, name: true } },
quality: true, integration: { select: { type: true } },
},
orderBy: { collectedAt: "desc" },
take: Number(limit ?? 100),
});
return rows.map((e) => ({
id: e.id, sequence: e.sequence, title: e.title,
controlSlug: e.control.slug, controlName: e.control.name,
sourceType: e.sourceType, integrationType: e.integration?.type ?? null,
collectedAt: e.collectedAt.toISOString(),
expiresAt: e.expiresAt?.toISOString() ?? null,
contentHash: e.contentHash, chainHash: e.chainHash,
quality: e.quality ? { score: e.quality.score, grade: e.quality.grade } : null,
}));
});
});
// ── TESTS LIST ───────────────────────────────────────────────────
app.get("/v1/tests", async (req) =>
rlsClient().asOrg(req.org!.id, async (tx) => {
const tests = await tx.controlTest.findMany({
include: {
control: { select: { slug: true, name: true } },
runs: { orderBy: { startedAt: "desc" }, take: 30 },
},
});
return tests.map((t) => ({
id: t.id, checkId: (t.check as { checkId?: string }).checkId ?? t.id,
name: t.name, schedule: t.schedule, enabled: t.enabled,
controlSlug: t.control.slug, controlName: t.control.name,
lastRunAt: t.lastRunAt?.toISOString() ?? null,
state: controlTestState(t.runs, 90),
recentRuns: t.runs.slice(0, 5).map((r) =>
({ status: r.status, at: r.startedAt.toISOString() })),
}));
}));
// ── RISK REGISTER ────────────────────────────────────────────────
app.get("/v1/risks", async (req) =>
rlsClient().asOrg(req.org!.id, async (tx) => {
const risks = await tx.risk.findMany({
include: { owner: { include: { user: { select: { name: true } } } },
controls: { include: { control: { select: { slug: true, name: true } } } } },
orderBy: [{ residualImpact: "desc" }, { createdAt: "desc" }],
});
return risks.map((r) => ({
id: r.id, title: r.title, category: r.category,
inherent: r.inherentLikelihood * r.inherentImpact,
residual: (r.residualLikelihood ?? r.inherentLikelihood) *
(r.residualImpact ?? r.inherentImpact),
treatment: r.treatment,
owner: r.owner?.user.name ?? null,
nextReviewAt: r.nextReviewAt?.toISOString() ?? null,
linkedControls: r.controls.map((c) => c.control),
}));
}));
// ── INTEGRATIONS: connected + connector catalog + processing ─────
app.get("/v1/integrations", async (req) => {
const connected = await rlsClient().asOrg(req.org!.id, async (tx) =>
tx.integration.findMany({
select: { id: true, type: true, status: true, lastSyncAt: true,
syncError: true, config: true },
}));
// Connector catalog — what each connector collects & which controls it feeds
const catalog: { type: string; specs: { dataType: string; title: string;
freshnessDays: number; controls: string[]; expectStable: boolean }[] }[] = [];
for (const type of CONNECTOR_TYPES) {
try {
const specs = connectorFor(type).specs();
catalog.push({ type, specs: specs.map((s) => ({
dataType: s.dataType, title: s.title, freshnessDays: s.freshnessDays,
controls: s.controlTemplateIds, expectStable: s.expectStable ?? false })) });
} catch { /* connector not registered */ }
}
return { connected, catalog,
processing: [
"1. connector.collect(dataType) fetches a normalized snapshot",
"2. contentHash = sha256(canonical items) — drift-detection key",
"3. Diff vs last IntegrationSnapshot; unchanged → skip (no evidence churn)",
"4. Changed → appendEvidence() into the control's hash chain (single tx)",
"5. If expectStable && changed → create a drift review task",
] };
});
// ── 50-STATE BREACH MATRIX (read) ────────────────────────────────
app.get("/v1/breach-matrix", async (req) => {
const { verification, jurisdiction } = req.query as
{ verification?: string; jurisdiction?: string };
const rules = await prisma.stateBreachRule.findMany({
where: {
...(verification ? { verificationStatus: verification } : {}),
...(jurisdiction ? { id: jurisdiction } : {}),
},
orderBy: { id: "asc" },
});
return rules.map((r) => ({
code: r.id, name: r.name, citation: r.citation,
deadlineDays: r.deadlineDays, notifyResidents: r.notifyResidents,
agThreshold: r.agThreshold, creditAgencyThreshold: r.creditAgencyThreshold,
otherAgencies: r.otherAgencies, harmGate: r.harmGate,
encryptionSafeHarbor: r.encryptionSafeHarbor,
verificationStatus: r.verificationStatus,
compiledConfidence: r.compiledConfidence, version: r.version,
lastVerifiedAt: r.lastVerifiedAt?.toISOString() ?? null,
}));
});
// ── PROPOSE A RULE AMENDMENT (manual correction) ─────────────────
// Creates a MatrixAmendmentProposal anchored to a synthetic "manual" bill,
// so it flows through the SAME MATRIX_PROPOSAL review queue as legislative
// amendments. Experts approve in the review console; never edit the rule row.
app.post("/v1/breach-matrix/:code/propose", async (req, reply) => {
const { code } = req.params as { code: string };
const { changes, effectiveAt, note } = req.body as {
changes: { field: string; to: unknown }[]; effectiveAt?: string; note?: string };
const rule = await prisma.stateBreachRule.findUnique({ where: { id: code } });
if (!rule) return reply.code(404).send({ title: "Rule not found", status: 404 });
if (!changes?.length)
return reply.code(400).send({ title: "No changes provided", status: 400 });
const ts = Date.now();
const bill = await prisma.trackedBill.create({ data: {
sourceId: "MANUAL", stateCode: code, billNumber: "MANUAL-AMENDMENT",
session: `manual-${ts}`, externalRef: `manual:${code}:${ts}`,
title: `Manual amendment proposal for ${code}`,
relevanceScore: 1, relevancePath: "MANUAL", stage: "ENACTED", status: "PROPOSED",
} });
const allowed = new Set(["deadlineDays", "notifyResidents", "agThreshold",
"creditAgencyThreshold", "otherAgencies", "harmGate",
"encryptionSafeHarbor", "substituteNoticeAllowed"]);
const clean = changes.filter((c) => allowed.has(c.field));
const current = { deadlineDays: rule.deadlineDays, notifyResidents: rule.notifyResidents,
agThreshold: rule.agThreshold, creditAgencyThreshold: rule.creditAgencyThreshold,
otherAgencies: rule.otherAgencies, harmGate: rule.harmGate,
encryptionSafeHarbor: rule.encryptionSafeHarbor };
const proposed = { ...current };
for (const c of clean) (proposed as Record<string, unknown>)[c.field] = c.to;
const proposal = await prisma.matrixAmendmentProposal.create({ data: {
billId: bill.id, stateRuleId: rule.id,
proposed: proposed as object,
diff: clean.map((c) => ({ field: c.field,
from: (current as Record<string, unknown>)[c.field], to: c.to })) as object,
effectiveAt: effectiveAt ? new Date(effectiveAt) : null,
extractorConfidence: 1.0, reviewerNote: note ?? "Manual amendment",
} });
return reply.code(201).send({ proposalId: proposal.id, status: proposal.status,
note: "Queued for expert review (MATRIX_PROPOSAL). The rule updates only after approval." });
});
// ── CONTROL FRAMEWORK: requirements + verification + mappings ────
app.get("/v1/framework/requirements", async (req) => {
const { framework } = req.query as { framework?: string };
const reqs = await prisma.frameworkRequirement.findMany({
where: framework ? { framework: { code: framework } } : {},
include: {
framework: { select: { code: true, name: true, isPublished: true } },
_count: { select: { controlMappings: true } },
},
orderBy: [{ frameworkId: "asc" }, { code: "asc" }],
});
return reqs.map((r) => ({
id: r.id, framework: r.framework.code, frameworkName: r.framework.name,
frameworkPublished: r.framework.isPublished,
code: r.code, title: r.title, category: r.category, kind: r.kind,
satisfiedByWorkflow: r.satisfiedByWorkflow,
verified: r.verified, verifiedBy: r.verifiedBy,
controlMappingCount: r._count.controlMappings,
}));
});
app.get("/v1/framework/templates", async () => {
const templates = await prisma.controlTemplate.findMany({ orderBy: { id: "asc" } });
return templates.map((t) => ({
id: t.id, name: t.name, category: t.category, description: t.description,
satisfiedBy: t.satisfiedBy, checkIds: t.checkIds, policyTypes: t.policyTypes,
mappings: (t.mappings as { framework: string; code: string; confidence: number }[]),
}));
});
// Propose mappings when an org adds a framework — expert accepts/rejects.
app.post("/v1/framework/mappings/suggest", async (req) => {
const { frameworkId } = req.body as { frameworkId: string };
return suggestMappingsForNewFramework(req.org!.id, frameworkId);
});
// ── REGULATORY RADAR: changes + sources + proposals ─────────────
app.get("/v1/radar/changes", async (req) => {
const { status, limit } = req.query as { status?: string; limit?: string };
const changes = await prisma.regChange.findMany({
where: status ? { status: status as never } : {},
include: {
sourceDocument: { select: { title: true, url: true, docType: true } },
impacts: { include: {
requirement: { select: { code: true, title: true } },
framework: { select: { code: true } } } },
},
orderBy: { createdAt: "desc" }, take: Number(limit ?? 50),
});
return changes.map((c) => ({
id: c.id, changeType: c.changeType, jurisdiction: c.jurisdiction,
summary: c.summary, effectiveAt: c.effectiveAt?.toISOString() ?? null,
status: c.status, extracted: c.extracted,
source: { title: c.sourceDocument.title, url: c.sourceDocument.url,
docType: c.sourceDocument.docType },
impacts: c.impacts.map((i) => ({
framework: i.framework.code, requirement: i.requirement?.code ?? null,
requirementTitle: i.requirement?.title ?? null,
impactLevel: i.impactLevel, rationale: i.rationale,
suggestedAction: i.suggestedAction,
})),
}));
});
app.get("/v1/radar/sources", async () => {
const [reg, legi] = await Promise.all([
prisma.regSource.findMany({ orderBy: { name: "asc" } }),
prisma.legislativeSource.findMany({ orderBy: { stateCode: "asc" } }),
]);
return {
federal: reg.map((s) => ({ name: s.name, authority: s.authority,
jurisdiction: s.jurisdiction, parser: s.parserType, enabled: s.enabled,
lastCheckedAt: s.lastCheckedAt?.toISOString() ?? null })),
state: legi.map((s) => ({ stateCode: s.stateCode, strategy: s.strategy,
sessionCycle: s.sessionCycle, watchSections: s.watchSections,
enabled: s.enabled, lastPolledAt: s.lastPolledAt?.toISOString() ?? null })),
};
});
app.get("/v1/radar/proposals", async () => {
const proposals = await prisma.matrixAmendmentProposal.findMany({
include: { bill: { select: { billNumber: true, stateCode: true, title: true } },
stateRule: { select: { name: true, citation: true } } },
orderBy: { createdAt: "desc" }, take: 50,
});
return proposals.map((p) => ({
id: p.id, status: p.status,
state: p.bill.stateCode, rule: p.stateRule.name,
bill: p.bill.billNumber, billTitle: p.bill.title,
changes: p.diff, effectiveAt: p.effectiveAt?.toISOString() ?? null,
confidence: p.extractorConfidence, createdAt: p.createdAt.toISOString(),
}));
});
}

await app.register(brainRoutes);
await app.register(brainStreamRoutes);
await app.register(auditorRoutes);
await app.register(auditorGdprRoutes);
await app.register(trustRoutes);
await app.register(reviewRoutes);
await app.register(ethicsEventsRoutes);
app.addHook("onError", async (req, reply, err) => {
emit({ type: "api.request.failed", source: req.url.startsWith("/mcp") ? "mcp" : "api",
orgId: req.org?.id, severityOverride: "WARN",
payload: { url: req.url, status: reply.statusCode,
error: String(err.message).slice(0, 200) } });
});
const port = Number(process.env.PORT ?? 3000);
await app.listen({ port, host: "0.0.0.0" });
console.log(`API on :${port}`);

process.on(sig, async () => { await app.close(); process.exit(0); });
}

import { useState } from "react";

collectedAt: string; expiresAt: string | null; contentHash: string; chainHash: string;

<div className="flex items-baseline justify-between">

))}

"use client";
import { useState } from "react";
import { api } from "../lib/api";
import { useApi } from "../lib/useApi";
import { relTime } from "../lib/format";
import { Badge, Button, Card, CardTitle, Skeleton } from "./ui";
const STATE_TONE = { PASSING: "pass", FAILING: "fail", DEGRADED: "warn", STALE: "warn", UNTESTED: "zinc" } as const;
interface Test { id: string; checkId: string; name: string; schedule: string; enabled: boolean;
controlSlug: string; controlName: string; lastRunAt: string | null; state: string;
recentRuns: { status: string; at: string }[] }
export function TestsClient() {
const { data, loading, refresh } = useApi<Test[]>("/v1/tests");
const [running, setRunning] = useState<string | null>(null);
const [flash, setFlash] = useState<string | null>(null);
const run = async (t: Test) => {
setRunning(t.checkId);
try {
const r = await api.runCheck(t.checkId);
setFlash(`Run started for ${t.name} → ${r.status}. Watch the live feed.`);
setTimeout(() => { setFlash(null); refresh(); }, 2500);
} catch (e) { setFlash(`Failed: ${(e as Error).message}`); }
finally { setRunning(null); }
};
if (loading) return <Skeleton className="h-96" />;
return (
<div className="mx-auto max-w-4xl space-y-4">
<div className="flex items-baseline justify-between">
<h1 className="text-lg font-semibold">Control Tests</h1>
<span className="text-xs text-muted-foreground">{(data ?? []).filter((t) => t.enabled).length} enabled</span>
</div>
{flash && <div className="rounded-md border border-info/30 bg-info/5 px-3 py-2 text-xs text-info">{flash}</div>}
<Card>
<CardTitle>Automated checks (predicate DSL)</CardTitle>
<div className="divide-y divide-border">
{(data ?? []).map((t) => (
<div key={t.id} className="flex items-center gap-3 px-4 py-3">
<span className={`h-2 w-2 rounded-full ${t.enabled ? "bg-pass" : "bg-zinc-300"}`} />
<div className="min-w-0 flex-1">
<p className="truncate text-[13px] font-medium">{t.name}</p>
<p className="text-xs text-muted-foreground">{t.controlName} · cron {t.schedule}{t.lastRunAt ? ` · last ${relTime(t.lastRunAt)}` : " · never run"}</p>
</div>
<div className="flex items-center gap-1">
{t.recentRuns.slice(0, 5).map((r, i) => (
<span key={i} title={`${r.status} ${relTime(r.at)}`}
className={`h-2 w-2 rounded-full ${r.status === "PASS" ? "bg-pass" : r.status === "FAIL" ? "bg-fail" : "bg-warn"}`} />
))}
</div>
<Badge tone={STATE_TONE[t.state as keyof typeof STATE_TONE] ?? "zinc"}>{t.state}</Badge>
<Button variant="outline" onClick={() => run(t)} disabled={running === t.checkId || !t.enabled}>
{running === t.checkId ? "…" : "Run now"}
</Button>
</div>
))}

## The end-to-end regulatory-change loop you now have in the UI

1. **`/radar` → Sources** shows the Federal Register feeds + every state legislature feed (auto-derived from the matrix).
2. **`/radar` → Changes** shows each detected `RegChange` with AI-graded `ChangeImpact`s (which framework/requirement, impact level, suggested action) and its review status.
3. **`/radar` → Matrix Proposals** + **`/matrix` → Propose amendment** let you modify rules — both create a `MatrixAmendmentProposal` that goes to the expert review queue; **nothing edits the rule row directly**.
4. **`/framework`** shows requirements (verified/unverified, CONTROL vs WORKFLOW) and control templates, so you can see exactly which requirements a reg change touches and how controls map across all four frameworks.
