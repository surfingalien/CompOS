import { FastifyInstance } from "fastify";
import { prisma } from "@db/client";
import { emit } from "@brain/emit";
/** Read-model for the Regulatory Intelligence Agent surfaces.
* All queries are defensive: if the batch-17 RIA tables aren't present yet
* the endpoints degrade to empty shapes instead of 500s. */
async function safe<T>(fn: () => Promise<T>, fallback: T): Promise<T> {
try { return await fn(); } catch { return fallback; }
}
export async function riaHandlers(app: FastifyInstance) {
// ── jurisdictions: crawl status + source rollups ──
app.get("/v1/ria/jurisdictions", async () => {
const jurisdictions = await safe(
() => prisma.jurisdiction.findMany({ orderBy: { code: "asc" } }), [] as any[]);
const sources = await safe(() => prisma.regSourceCountry.findMany(), [] as any[]);
const crawls = await safe(
() => prisma.crawlTask.findMany({ orderBy: { startedAt: "desc" } }), [] as any[]);
return jurisdictions.map((j: any) => {
const js = sources.filter((s: any) => s.jurisdictionId === j.id);
const jc = crawls.filter((c: any) => c.jurisdictionId === j.id);
const last = jc[0];
return {
id: j.id, code: j.code, name: j.name, region: j.region,
legalTradition: j.legalTradition,
sourceCount: js.length,
avgReliability: js.length
? js.reduce((a: number, s: any) => a + (s.reliability ?? 0.5), 0) / js.length
: null,
lastCrawlAt: last?.startedAt ?? null,
lastCrawlStatus: last?.status ?? null,
docsExtracted: jc.reduce((a: number, c: any) => a + (c.docsExtracted ?? 0), 0),
enabled: j.enabled,
};
});
});
// ── source reliability detail ──
app.get("/v1/ria/sources", async () => {
const sources = await safe(
() => prisma.regSourceCountry.findMany({ orderBy: { reliability: "desc" } }), [] as any[]);
const jurisdictions = await safe(() => prisma.jurisdiction.findMany(), [] as any[]);
const byId = new Map(jurisdictions.map((j: any) => [j.id, j.code]));
return sources.map((s: any) => ({
id: s.id, name: s.name, url: s.url, parserType: s.parserType,
jurisdictionCode: byId.get(s.jurisdictionId) ?? null,
reliability: s.reliability ?? 0.5,
status: s.status,
lastCheckedAt: s.lastCheckedAt,
docsExtracted: s.docsExtracted ?? 0,
crawlCount: s.crawlCount ?? 0,
discoveredBy: s.discoveredBy,
}));
});
// ── knowledge graph: reg → requirement → control ──
app.get("/v1/ria/graph", async (req) => {
const { jurisdiction } = req.query as { jurisdiction?: string };
const nodes = await safe(() => prisma.regKnowledgeNode.findMany({
where: jurisdiction ? { jurisdictionId: jurisdiction } : {},
orderBy: { confidence: "desc" }, take: 500,
}), [] as any[]);
const jurisdictions = await safe(() => prisma.jurisdiction.findMany(), [] as any[]);
const byId = new Map(jurisdictions.map((j: any) => [j.id, j.code]));
const counts = { REGULATION: 0, REQUIREMENT: 0, CONTROL: 0, GUIDANCE: 0 };
for (const n of nodes)
if ((counts as any)[n.nodeType] !== undefined) (counts as any)[n.nodeType]++;
return {
counts,
nodes: nodes.map((n: any) => ({
id: n.id, nodeType: n.nodeType, refCode: n.refCode, title: n.title,
jurisdictionCode: byId.get(n.jurisdictionId) ?? null,
confidence: n.confidence, verified: n.verified,
linkedRequirementId: n.linkedRequirementId,
linkedControlTemplateId: n.linkedControlTemplateId,
})),
};
});
// ── extraction drift: RIA-relevant open anomalies from the brain ──
app.get("/v1/ria/drift", async () => {
const { brainDb } = await import("@brain");
const anomalies = await safe(() => brainDb.anomaly.findMany({
where: { status: "OPEN", OR: [
{ subsystem: { contains: "ria", mode: "insensitive" } },
{ subsystem: { contains: "extract", mode: "insensitive" } },
{ subsystem: { contains: "crawl", mode: "insensitive" } },
{ subsystem: { contains: "retrieval", mode: "insensitive" } },
{ metric: { contains: "reject", mode: "insensitive" } },
{ metric: { contains: "drift", mode: "insensitive" } },
] },
orderBy: { at: "desc" }, take: 30,
}), [] as any[]);
return anomalies.map((a: any) => ({
id: a.id, detector: a.detector, subsystem: a.subsystem, metric: a.metric,
value: a.value, baseline: a.baseline, zScore: a.zScore,
severity: a.severity, at: a.at, details: a.details,
}));
});
// ── learning capture: wired from the review console on reject/modify ──
// Emits a brain event (the learning substrate) and persists a signal row
// when the RIA learning table exists. Feeds monitorRetrievalHealth/calibration.
app.post("/v1/ria/feedback", async (req) => {
const { reviewItemId, jurisdictionCode, changeType, decision, reasonCode, note } =
req.body as {
reviewItemId?: string; jurisdictionCode?: string; changeType?: string;
decision: "APPROVED" | "REJECTED" | "MODIFIED";
reasonCode?: string; note?: string };
emit({ type: "regchange.applied", source: "review",
payload: { kind: "ria_review_feedback", reviewItemId, jurisdictionCode,
changeType, decision, reasonCode, note } });
const stored = await safe(async () =>
await (prisma as any).regLearningSignal?.create({
data: { jurisdictionCode: jurisdictionCode ?? "UNKNOWN",
changeType: changeType ?? "UNKNOWN", decision,
reasonCode: reasonCode ?? null, note: note ?? null,
reviewItemId: reviewItemId ?? null },
}) ?? null, null);
return { captured: true, stored: !!stored };
});
}

async function req<T>(method: string, path: string, body?: unknown): Promise<T> {
const res = await fetch(`${API_BASE}${path}`, {
method,
headers: body ? { "content-type": "application/json" } : undefined,
body: body ? JSON.stringify(body) : undefined,
credentials: "include",
});
if (!res.ok) throw new ApiError(res.status, await res.json().catch(() => ({})));
return (res.status === 204 ? undefined : await res.json()) as T;
}

</div>
);

import type { ReviewBundleBlock, ReviewDetail, ReviewQueueItem } from "../lib/api";
import { useApi } from "../lib/useApi";
import { relTime } from "../lib/format";
import { Badge, Button, Card, Kbd, Modal, Skeleton } from "./ui";
const QUEUE_LABEL: Record<string, string> = {
FRAMEWORK_CONTENT: "Framework content", MATRIX_PROPOSAL: "Matrix proposal", REG_CHANGE: "Reg change" };
const PRIO_TONE: Record<string, "pass" | "fail" | "warn" | "info" | "zinc"> = {
CRITICAL: "fail", HIGH: "warn", MEDIUM: "info", LOW: "zinc" };
const STATUS_TONE: Record<string, "pass" | "fail" | "warn" | "info" | "zinc"> = {
PENDING: "warn", CLAIMED: "info", APPROVED_PENDING_SECOND: "info",
APPROVED: "pass", REJECTED: "fail", DEFERRED: "zinc", SUPERSEDED: "zinc" };
// Learning-capture taxonomy for rejections (feeds the RIA learning loop)
const REJECT_CODES = [
{ code: "NOT_A_CHANGE", label: "Not a regulatory change" },
{ code: "WRONG_JURISDICTION", label: "Wrong jurisdiction" },
{ code: "MISREAD_DEADLINE", label: "Misread the deadline" },
{ code: "OVER_EXTRACTED", label: "Over-extracted (invented obligations)" },
{ code: "UNDER_EXTRACTED", label: "Under-extracted (missed obligations)" },
{ code: "DUPLICATE", label: "Duplicate of an existing change" },
{ code: "OTHER", label: "Other" },
] as const;
export function ReviewConsole() {
const queue = useApi("/v1/review/queue", api.reviewQueue);
const stats = useApi("/v1/review/stats", api.reviewStats);
const [selectedId, setSelectedId] = useState<string | null>(null);
const [detail, setDetail] = useState<ReviewDetail | null>(null);
const [detailErr, setDetailErr] = useState<string | null>(null);
const [filter, setFilter] = useState<string | null>(null);
const [busy, setBusy] = useState(false);
const [flash, setFlash] = useState<string | null>(null);
const [note, setNote] = useState("");
const [rejectOpen, setRejectOpen] = useState(false);
const [reasonCode, setReasonCode] = useState<string>("OTHER");
const say = (m: string) => { setFlash(m); setTimeout(() => setFlash(null), 3500); };
const items = useMemo(() => {
const list = queue.data ?? [];
return filter ? list.filter((i) => i.queueType === filter) : list;
}, [queue.data, filter]);
const openDetail = async (id: string) => {
setSelectedId(id); setDetail(null); setDetailErr(null); setNote("");
try { setDetail(await api.reviewItem(id)); }
catch (e) {
setDetailErr(e instanceof ApiError && e.status === 403
? "Requires a platform reviewer credential." : (e as Error).message);
}
};
const doScan = async () => {
setBusy(true);
try { const r = await api.reviewScan(); say(`Scan complete — ${r.enqueued} item(s) enqueued`); queue.refresh(); }
catch (e) { say(`Scan failed: ${(e as Error).message}`); }
finally { setBusy(false); }
};
const doClaim = async () => {
if (!selectedId) return; setBusy(true);
try { await api.reviewClaim(selectedId); say("Claimed"); await openDetail(selectedId); queue.refresh(); }
catch (e) { say(`Claim failed: ${(e as Error).message}`); }
finally { setBusy(false); }
};
const doApprove = async () => {
if (!selectedId) return; setBusy(true);
try {
await api.reviewDecide(selectedId, "APPROVE", note);
// learning capture: positive signal
await api.riaFeedback({ reviewItemId: selectedId,
jurisdictionCode: detail?.item.jurisdiction ?? undefined,
decision: "APPROVED", note });
say("Approved — applied downstream (fan-out / matrix update)");
setSelectedId(null); setDetail(null); queue.refresh();
} catch (e) { say(`Approve failed: ${(e as Error).message}`); }
finally { setBusy(false); }
};
const doReject = async () => {
if (!selectedId) return; setBusy(true);
try {
await api.reviewDecide(selectedId, "REJECT", `[${reasonCode}] ${note}`);
// learning capture: rejection reason feeds extraction calibration
await api.riaFeedback({ reviewItemId: selectedId,
jurisdictionCode: detail?.item.jurisdiction ?? undefined,
changeType: detail?.item.queueType, decision: "REJECTED",
reasonCode, note });
say(`Rejected (${reasonCode}) — captured for the learning loop`);
setRejectOpen(false); setSelectedId(null); setDetail(null); queue.refresh();
} catch (e) { say(`Reject failed: ${(e as Error).message}`); }
finally { setBusy(false); }
};
const doRelease = async () => {
if (!selectedId) return; setBusy(true);
try { await api.reviewRelease(selectedId); say("Released"); setSelectedId(null); setDetail(null); queue.refresh(); }
catch (e) { say(`Release failed: ${(e as Error).message}`); }
finally { setBusy(false); }
};
const slaBreaches = stats.data?.slaBreaches ?? 0;
return (
<div className="flex h-[calc(100vh-3rem)] flex-col">
<div className="flex flex-wrap items-center gap-3 border-b border-border pb-3">
<h1 className="text-[15px] font-semibold">Review Console — Expert Gate</h1>
<Badge tone="zinc">{(queue.data ?? []).length} queued</Badge>
{slaBreaches > 0 && <Badge tone="fail">{slaBreaches} SLA breaches</Badge>}
<div className="flex-1" />
<Button variant="outline" onClick={doScan} disabled={busy}>{busy ? "Scanning…" : "Scan queues"}</Button>
</div>
{flash && <div className="border-b border-info/30 bg-info/5 px-4 py-1.5 text-xs text-info">{flash}</div>}
<div className="mt-2 flex flex-wrap gap-2">
<button onClick={() => setFilter(null)}
className={`rounded border px-2 py-1 text-xs ${filter === null ? "border-zinc-900 bg-zinc-900 text-white" : "border-border"}`}>All</button>
{Object.keys(QUEUE_LABEL).map((q) => (
<button key={q} onClick={() => setFilter(q)}
className={`rounded border px-2 py-1 text-xs ${filter === q ? "border-zinc-900 bg-zinc-900 text-white" : "border-border"}`}>
{QUEUE_LABEL[q]}</button>
))}
</div>
<div className="mt-3 grid flex-1 grid-cols-1 overflow-hidden lg:grid-cols-[360px_1fr]">
{/* LEFT — queue */}
<div className="overflow-y-auto border-r border-border">
{queue.loading ? <div className="p-4"><Skeleton /></div> : items.map((it) => (
<button key={it.id} onClick={() => openDetail(it.id)}
className={`flex w-full flex-col gap-1 border-b border-border px-3 py-2.5 text-left ${selectedId === it.id ? "bg-muted" : "hover:bg-muted/50"}`}>
<div className="flex items-center gap-2">
<Badge tone={PRIO_TONE[it.priority] ?? "zinc"}>{it.priority}</Badge>
<span className="min-w-0 flex-1 truncate text-[13px] font-medium">{it.title}</span>
</div>
<div className="flex items-center gap-2 text-[11px] text-muted-foreground">
<span>{QUEUE_LABEL[it.queueType] ?? it.queueType}</span>
{it.jurisdiction && <span className="font-mono">{it.jurisdiction}</span>}
<span>· conf {it.confidence.toFixed(2)}</span>
<span className="ml-auto">due {relTime(it.slaDueAt)}</span>
</div>
<div className="flex items-center gap-2">
<Badge tone={STATUS_TONE[it.status] ?? "zinc"}>{it.status}</Badge>
{it.requiredCredential && <Badge tone="draft">{it.requiredCredential}</Badge>}
</div>
</button>
))}
{items.length === 0 && !queue.loading && (
<p className="p-8 text-center text-xs text-muted-foreground">Queue is clear. Run a scan to enqueue new items.</p>
)}
</div>
{/* RIGHT — detail */}
<div className="overflow-y-auto p-5">
{!selectedId && <EmptyDetail />}
{selectedId && detailErr && (
<Card className="p-6 text-center text-sm text-muted-foreground">{detailErr}</Card>
)}
{selectedId && detail && (
<DetailPane item={detail.item} bundle={detail.bundle}
note={note} setNote={setNote} busy={busy}
onClaim={doClaim} onApprove={doApprove}
onReject={() => setRejectOpen(true)} onRelease={doRelease} />
)}
</div>
</div>
{/* reject modal with learning taxonomy */}
<Modal open={rejectOpen} onClose={() => setRejectOpen(false)} title="Reject — reason feeds the learning loop">
<div className="space-y-2">
{REJECT_CODES.map((c) => (
<label key={c.code} className="flex items-center gap-2 rounded-md border border-border px-3 py-2 text-[13px]">
<input type="radio" name="reason" checked={reasonCode === c.code}
onChange={() => setReasonCode(c.code)} />
{c.label}
</label>
))}
<textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2}
placeholder="Reviewer note (optional)…"
className="w-full rounded-md border border-border p-2 text-[13px]" />
<div className="flex justify-end gap-2">
<Button variant="ghost" onClick={() => setRejectOpen(false)}>Cancel</Button>
<Button variant="danger" onClick={doReject} disabled={busy}>{busy ? "Rejecting…" : "Reject"}</Button>
</div>
</div>
</Modal>
</div>
);
}
function EmptyDetail() {
return (
<div className="flex h-full flex-col items-center justify-center text-center">
<p className="text-sm font-medium">Select an item to review</p>
<p className="mt-1 max-w-sm text-xs text-muted-foreground">
Approving applies the change downstream (fan-out tasks / matrix update / framework publish).
Rejecting captures a reason code that calibrates the extraction pipeline.</p>
</div>
);
}
function DetailPane({ item, bundle, note, setNote, busy, onClaim, onApprove, onReject, onRelease }: {
item: ReviewQueueItem; bundle: ReviewDetail["bundle"]; note: string;
setNote: (s: string) => void; busy: boolean;
onClaim: () => void; onApprove: () => void; onReject: () => void; onRelease: () => void;
}) {
const claimed = item.status === "CLAIMED" || item.status === "APPROVED_PENDING_SECOND";
const needsSecond = item.status === "APPROVED_PENDING_SECOND";
return (
<div className="space-y-4">
<div>
<div className="flex items-center gap-2">
<Badge tone={PRIO_TONE[item.priority] ?? "zinc"}>{item.priority}</Badge>
<Badge tone="zinc">{QUEUE_LABEL[item.queueType] ?? item.queueType}</Badge>
{item.jurisdiction && <Badge tone="info">{item.jurisdiction}</Badge>}
{item.requiredCredential && <Badge tone="draft">requires {item.requiredCredential}</Badge>}
</div>
<h2 className="mt-2 text-[15px] font-semibold leading-relaxed">{bundle.header.title}</h2>
<p className="mt-1 text-xs text-muted-foreground">
Source: {bundle.header.source}{bundle.header.link && <> · <a href={bundle.header.link} target="_blank" rel="noreferrer" className="text-info hover:underline">open ↗</a></>}
</p>
</div>
{bundle.risk.length > 0 && (
<div className="rounded-md border border-warn/30 bg-warn/5 p-3">
<p className="text-xs font-medium text-warn">Reviewer caution</p>
<ul className="mt-1 list-disc pl-4 text-xs text-zinc-700">
{bundle.risk.map((r, i) => <li key={i}>{r}</li>)}
</ul>
</div>
)}
<div className="space-y-3">
{bundle.body.map((block, i) => <BundleBlock key={i} block={block} />)}
</div>
<textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2}
placeholder="Decision note (recorded in the audit trail)…"
className="w-full rounded-md border border-border p-2 text-[13px]" />
<div className="flex flex-wrap gap-2">
{!claimed && <Button variant="outline" onClick={onClaim} disabled={busy}>Claim</Button>}
{claimed && <Button variant="ghost" onClick={onRelease} disabled={busy}>Release</Button>}
<div className="flex-1" />
<Button variant="danger" onClick={onReject} disabled={busy}>Reject</Button>
<Button variant="success" onClick={onApprove} disabled={busy}>
{needsSecond ? "Second approval (four-eyes)" : "Approve"} <Kbd>A</Kbd>
</Button>
</div>
{item.priority === "CRITICAL" && !needsSecond && (
<p className="text-[11px] text-muted-foreground">CRITICAL item — approval requires a second reviewer (four-eyes).</p>
)}
</div>
);
}
function BundleBlock({ block }: { block: ReviewBundleBlock }) {
switch (block.kind) {
case "diff-table": {
const rows = block.rows as { field: string; from: string; to: string; quote?: string; confidence?: number }[];
return (
<Card><CardTitle>Proposed change (diff)</CardTitle>
<div className="overflow-x-auto"><table className="w-full text-[13px]">
<thead><tr className="border-b border-border text-left text-[11px] uppercase tracking-wide text-muted-foreground">
<th className="px-3 py-2">Field</th><th className="px-3 py-2">From</th><th className="px-3 py-2">To</th><th className="px-3 py-2">Conf</th></tr></thead>
<tbody className="divide-y divide-border">{rows.map((r, i) => (
<tr key={i}>
<td className="px-3 py-2 font-mono text-xs">{r.field}</td>
<td className="px-3 py-2 text-xs text-fail line-through">{r.from}</td>
<td className="px-3 py-2 text-xs text-pass">{r.to}</td>
<td className="px-3 py-2 text-xs tabular-nums">{r.confidence?.toFixed(2) ?? "—"}</td>
</tr>))}</tbody></table></div>
</Card>
);
}
case "paraphrase-card": {
const req = block.requirement as { code: string; title: string; desc: string };
return (
<Card><CardTitle>Requirement paraphrase <span className="normal-case text-[10px] text-muted-foreground">(verify against licensed source)</span></CardTitle>
<div className="p-3">
<p className="font-mono text-xs text-info">{req.code}</p>
<p className="mt-1 text-[13px] font-medium">{req.title}</p>
<p className="mt-1 text-xs leading-relaxed text-muted-foreground">{req.desc}</p>
{block.officialHint && <p className="mt-2 text-[11px] text-warn">{String(block.officialHint)}</p>}
</div>
</Card>
);
}
case "impact-list": {
const items = block.items as { label: string; level: string; action?: string }[];
return (
<Card><CardTitle>Impact assessment</CardTitle>
<div className="divide-y divide-border">{items.map((it, i) => (
<div key={i} className="flex items-center gap-2 px-3 py-2 text-[13px]">
<Badge tone={it.level === "CRITICAL" ? "fail" : it.level === "MAJOR" ? "warn" : it.level === "MINOR" ? "info" : "zinc"}>{it.level}</Badge>
<span className="min-w-0 flex-1">{it.label}</span>
{it.action && <span className="text-xs text-muted-foreground">{it.action}</span>}
</div>))}</div>
</Card>
);
}
case "evidence-quote": {
const quotes = block.quotes as { source: string; text: string }[];
return (
<Card><CardTitle>Extracted evidence</CardTitle>
<div className="space-y-2 p-3">{quotes.map((q, i) => (
<div key={i} className="border-l-2 border-border pl-3">
<p className="text-[11px] font-medium text-muted-foreground">{q.source}</p>
<p className="mt-0.5 text-xs leading-relaxed text-zinc-700">{q.text}</p>
</div>))}</div>
</Card>
);
}
case "affected-orgs": {
const count = block.count as number; const orgs = block.orgs as string[];
return (
<Card><CardTitle>Blast radius</CardTitle>
<div className="p-3 text-[13px]">
<p><span className="font-semibold tabular-nums">{count}</span> org(s) affected via accepted mappings.</p>
{orgs.length > 0 && <p className="mt-1 font-mono text-[11px] text-muted-foreground">{orgs.join(", ")}{count > orgs.length ? " …" : ""}</p>}
</div>
</Card>
);
}
default:
return null;
}
}
